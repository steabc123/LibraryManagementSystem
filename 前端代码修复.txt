import request from '@/utils/request'

// 获取购物车列表
export function getCart() {
    return request({
        url: '/cart/list',
        method: 'get'
    })
}

// 添加图书到购物车
export function addToCart(bookData) {
    return request({
        url: '/cart',
        method: 'post',
        data: {
            action: 'add',
            ...bookData
        }
    })
}

// 从购物车移除图书
export function removeFromCart(cartId) {
    return request({
        url: '/cart/' + cartId,
        method: 'delete'
    })
}

// 清空购物车
export function clearCart() {
    return request({
        url: '/cart',
        method: 'post',
        data: {
            action: 'clear'
        }
    })
}

// 更新购物车商品数量
export function updateCartQuantity(cartItemId, quantity) {
    return request({
        url: '/cart',
        method: 'put',
        data: {
            cartItemId,
            quantity
        }
    })
}

// 获取购物车总金额
export function getCartTotal() {
    return request({
        url: '/cart/total',
        method: 'get'
    })
}

async handleAddToCart(book) {

    await addToCart({

      quantity: 1
    })
    alert('已加入购物车')
  } catch (error) {
    alert('加入购物车失败：' + error.message)
  }
},

async loadCart() {
  try {
    const res = await getCart()
    this.cartItems = res.data || []
    
    // 计算总金额
    const totalRes = await getCartTotal()
    if (totalRes && totalRes.data) {
      this.totalAmount = totalRes.data.totalAmount
    } else {
      // 如果接口失败，手动计算
      this.totalAmount = this.cartItems.reduce((sum, item) => {
        return sum + ((item.price || 0) * (item.quantity || 1))
      }, 0)
    }
  } catch (error) {
    alert('加载购物车失败：' + error.message)
  }
},

// 5. 添加更新数量的方法（在methods中添加）

    await updateCartQuantity(cartItemId, parseInt(quantity))
    this.loadCart()
  } catch (error) {
    alert('更新数量失败：' + error.message)
  }
},
